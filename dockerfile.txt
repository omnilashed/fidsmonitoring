# Menggunakan PHP 8.0 sesuai dengan xampp Anda
FROM php:8.0-apache

# Mengaktifkan mod_rewrite Apache (diperlukan untuk Framework/Routing)
RUN a2enmod rewrite

# Menginstal ekstensi PHP standar yang biasa dipakai XAMPP (mysqli, pdo_mysql, gd, zip)
RUN apt-get update && apt-get install -y \
    libpng-dev \
    libjpeg-dev \
    libfreetype6-dev \
    libzip-dev \
    zip \
    && docker-php-ext-configure gd --with-freetype --with-jpeg \
    && docker-php-ext-install -j$(nproc) gd mysqli pdo pdo_mysql zip

# Menyalin seluruh source code ke direktori web server
COPY . /var/www/html/

# Mengatur hak akses folder
RUN chown -R www-data:www-data /var/www/html

EXPOSE 80
